Replace the full contents of `frontend/src/App.jsx` with:

```jsx
import { useEffect, useState } from 'react'
import Leaderboard from './components/Leaderboard.jsx'
import ThisWeekSidebar from './components/ThisWeekSidebar.jsx'
import { api } from './api.js'

export default function App() {
  const [rows, setRows] = useState(null)
  const [error, setError] = useState(null)

  useEffect(() => {
    api.leaderboard().then(setRows).catch((e) => setError(e.message))
  }, [])

  return (
    <div className="app">
      <header className="header">
        <span className="header-mark" />
        <div>
          <h1>Fantasy Movie League</h1>
          <p>Season standings · click any round for full breakdown</p>
        </div>
      </header>

      {error && (
        <div className="state-msg">
          Couldn't reach the backend — confirm the API is running on :8000 and refresh.
        </div>
      )}
      {!error && !rows && <div className="state-msg">Loading…</div>}
      {!error && rows && rows.length === 0 && (
        <div className="empty-state">
          <div className="empty-state-heading">No standings yet</div>
          <div className="empty-state-body">Add players and movie picks to the league data to see rankings here.</div>
        </div>
      )}
      {rows && rows.length > 0 && (
        <div className="app-layout">
          <main className="app-main">
            <Leaderboard rows={rows} />
          </main>
          <ThisWeekSidebar />
        </div>
      )}
    </div>
  )
}
```

Notes on decisions baked into this code (do not deviate):
- The top-level mockup nav bar (League / My Team / Titles / Trades / Calendar links, "LINEUP LOCKS" meta) is intentionally OMITTED — this app has no routing (`ARCHITECTURE.md`: "No frontend routing library — the entire UI is a single page") and Component & Data Mapping's scope principle is to omit every fictional-only element with no real backing functionality; a nav bar linking to nonexistent pages would be exactly that.
- `.header h1` keeps the existing "Fantasy Movie League" text (not renamed to the mockup's fictional "Movie League") — UI-SPEC's Copywriting Contract does not define new app-title copy, only empty/error/CTA copy, so the existing title text is preserved and only its typography changes (via CSS from Plan 01-01).
- Error copy is the exact locked string: "Couldn't reach the backend — confirm the API is running on :8000 and refresh."
- Empty-state copy is exact and only shown when `rows` is a non-null empty array (edge case per UI-SPEC, not reachable with current seed data).
