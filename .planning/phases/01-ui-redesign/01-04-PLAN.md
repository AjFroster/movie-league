```jsx
import { useState } from 'react'

function statusFor(m) {
  if (!m.movie) return { dot: 'no-pick', label: 'NO PICK' }
  if (m.penalties !== 0) return { dot: 'at-risk', label: 'AT RISK' }
  if (m.imdb === null && m.gross === null && m.rt_crit === null) return { dot: 'awaiting', label: 'AWAITING SCORE' }
  return { dot: 'scored', label: 'SCORED' }
}

function MovieDetail({ m, ownerCount }) {
  // Placeholder — Task 2 replaces this body with the full expanded panel
  // (hero, meta, stat strip, points ledger, campaign tracker, ownership callout).
  return (
    <div className="movie-detail">
      <div>{m.total} pts</div>
    </div>
  )
}

export default function MovieCard({ movie: m, ownerCount }) {
  const [open, setOpen] = useState(false)
  const isPending = m.imdb === null && m.gross === null && m.rt_crit === null
  const status = statusFor(m)

  function handleToggle() {
    if (!isPending) setOpen(o => !o)
  }

  return (
    <>
      <div
        className={`roster-row${isPending ? ' pending' : ''}${open ? ' expanded' : ''}`}
        onClick={handleToggle}
        role={isPending ? undefined : 'button'}
        tabIndex={isPending ? undefined : 0}
        aria-expanded={isPending ? undefined : open}
        onKeyDown={e => {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault()
            handleToggle()
          }
        }}
      >
        <span className="roster-col-round">R{m.round}</span>
        <span className={`roster-col-title${!m.movie ? ' dim' : ''}`}>{m.movie || 'Not yet picked'}</span>
        <span className="status-pill">
          <span className={`status-dot ${status.dot}`} />
          <span className="status-label">{status.label}</span>
        </span>
        <span className={`roster-col-pts ${m.total > 0 ? 'positive' : m.total < 0 ? 'negative' : 'zero'}`}>
          {!m.movie ? '—' : (m.total > 0 ? `+${m.total}` : m.total)}
        </span>
        <span className="roster-col-watched">{m.who_watched.length}/{ownerCount}</span>
      </div>
      {open && <MovieDetail m={m} ownerCount={ownerCount} />}
    </>
  )
}
```

Key decisions baked into this code (do not deviate):
- `isPending` and `handleToggle` logic is BYTE-IDENTICAL to the pre-redesign version (`m.imdb === null && m.gross === null && m.rt_crit === null`, disables click) — this is the existing behavior UI-04 requires to be preserved, not a new design.
- Precedence reconciliation (settle this here, do NOT reorder on your own): UI-SPEC's Status Pill Token Table lists "No pick" with condition `!m.movie`, and its prose adds "penalty check first, then pending/awaiting checks, else scored." The order written above — no-pick, at-risk, awaiting, scored — satisfies both: `!m.movie` is a data-shape guard that must run first (a round with no title cannot meaningfully be "at risk"), and among the remaining three states the penalty check does come before the awaiting/scored checks exactly as the prose requires. With current data this ordering difference is unreachable anyway (`movie` is a required non-empty string on all 30 entries).
- AWAITING SCORE and `isPending` share the identical condition by design: an awaiting row is always non-clickable, and a clickable row is never AWAITING SCORE. Do not attempt to create an "awaiting but clickable" row — it is not a reachable state.
- The row and its expanded detail are siblings inside a Fragment (`<>...</>`), not nested — `.movie-detail` renders as a full-width block below `.roster-row` inside PlayerCard's `.roster-table` (a flex column), matching "renders inline as a full-width expanded row beneath the clicked table row."
- WATCHED column shows `{m.who_watched.length}/{ownerCount}` (e.g. "3/5") — `ownerCount` comes from the prop, never hardcoded.
