Replace the full contents of `frontend/src/components/PlayerCard.jsx` with:

```jsx
import { useEffect, useState } from 'react'
import MovieCard from './MovieCard.jsx'
import { api } from '../api.js'

const ORDINALS = ['', '1ST', '2ND', '3RD', '4TH', '5TH', '6TH', '7TH', '8TH', '9TH', '10TH']

function ordinal(n) {
  return ORDINALS[n] || `${n}TH`
}

function signed(v) {
  return v > 0 ? `+${v}` : `${v}`
}

export default function PlayerCard({ summary, ownerCount }) {
  const [movies, setMovies] = useState(null)
  const [error, setError] = useState(null)

  useEffect(() => {
    api.owner(summary.owner)
      .then(d => setMovies(d.movies))
      .catch(e => setError(e.message))
  }, [summary.owner])

  return (
    <div className={`player-card${summary.rank === 1 ? ' rank-1' : ''}`}>
      <div className="player-header">
        <div className="player-rank">#{summary.rank}</div>
        <div className="player-identity">
          <div className="player-meta-line">
            {ordinal(summary.rank)} OF {ownerCount} · {summary.rounds_played} ROUND{summary.rounds_played === 1 ? '' : 'S'}
          </div>
          <div className="player-name">{summary.owner}</div>
        </div>
        <div className="player-stats">
          <div className="stat-block">
            <span className="stat-block-label">TOTAL</span>
            <span className={`stat-block-value ${summary.total > 0 ? 'positive' : summary.total < 0 ? 'negative' : ''}`}>
              {signed(summary.total)}
            </span>
          </div>
          <div className="stat-block">
            <span className="stat-block-label">RATING</span>
            <span className="stat-block-value">{signed(summary.rating_score)}</span>
          </div>
          <div className="stat-block">
            <span className="stat-block-label">FINANCIAL</span>
            <span className="stat-block-value">{signed(summary.financial_score)}</span>
          </div>
          <div className="stat-block">
            <span className="stat-block-label">WATCH</span>
            <span className="stat-block-value">+{summary.watch_points}</span>
          </div>
          {summary.penalties !== 0 && (
            <div className="stat-block">
              <span className="stat-block-label">PENALTIES</span>
              <span className="stat-block-value negative">{signed(summary.penalties)}</span>
            </div>
          )}
        </div>
      </div>

      <div className="roster-table">
        <div className="roster-header-row">
          <span>ROUND</span>
          <span>TITLE</span>
          <span>STATUS</span>
          <span>PTS</span>
          <span>WATCHED</span>
        </div>
        {error && <div className="card-error" role="alert">{error}</div>}
        {!movies && !error && <div className="state-msg">Loading rounds…</div>}
        {movies && movies.map(m => (
          <MovieCard key={m.round} movie={m} ownerCount={ownerCount} />
        ))}
      </div>
    </div>
  )
}
```

Key decisions baked into this code (do not deviate):
- SLOT is relabeled ROUND, PROJ is omitted entirely, OWNED is relabeled WATCHED, per the locked Component & Data Mapping table (no "PROJ" text anywhere in this file).
- The PENALTIES stat block is REQUIRED whenever `summary.penalties !== 0`. Without it the four remaining blocks visibly contradict TOTAL against real data (e.g. RATING 72 + FIN 11 + WATCH 10 = 93 shown beside TOTAL 43, because penalties are −50), which breaks the project's core value of accurate score breakdowns. The pre-redesign UI carried a `⚠ Penalty` chip for exactly this reason — this block is its replacement, not a new feature.
- The meta line implements UI-SPEC's Typography example ("2ND OF 5 · 6 ROUNDS") using `summary.rank`, the `ownerCount` prop, and `summary.rounds_played`. All three are real fields already present in the `GET /api/leaderboard` response — nothing is fabricated.
- Rank-1 gets `className="player-card rank-1"` (flat amber top-border via CSS from Plan 01-01); ranks 2+ get no rank modifier class and no per-rank color — matches "no glow, no gradient" requirement.
- The old `RANK_COLORS` map, `MAX_SCORE` progress bar, `ScoreChip` emoji component, and inline `style={{ '--rank-color': ... }}` are all removed — none of that survives in the new design.
